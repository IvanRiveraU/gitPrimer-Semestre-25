	\noindent\textbf{Instrucciones:} Realice la división larga indicando el cociente y el residuo.
	
	\begin{enumerate}[label=(\arabic*)]
		\item Divida $\;x^{3}-8\;$ entre $\;x-2.$
		\item Divida $\;2x^{4}-3x^{3}+0x^{2}+5x-6\;$ entre $\;x^{2}-x+2.$
		\item Divida $\;4x^{5}+0x^{4}+x^{2}-7\;$ entre $\;2x^{2}+1.$
		\item Divida $\;-x^{4}+6x^{2}-9x+4\;$ entre $\;-x^{2}+3x-2.$
	\end{enumerate}
	
	\bigskip
	

	\noindent\textbf{Instrucciones:} Utilice solo el teorema del residuo para evaluar f(x) (Dejando indicado cual es el residuo)
	
	\begin{enumerate}[label=(\arabic*), start=5]
		\item Para $\;f(x)=2x^{4}-5x^{3}+3x-7\;$ encuentre el residuo al dividir entre $\;x+3.$
		\item Para $\;f(x)=-x^{5}+4x^{2}-1\;$ encuentre el residuo al dividir entre $\;x-2.$
	\end{enumerate}
	
		\bigskip
	
	\noindent\textbf{Instrucciones:} Verifique si el binomio dado es factor de la función f(x) (Dejando indicado si es factor o no)
	
	\begin{enumerate}[label=(\arabic*), start=7]
		\item Verifique si $\;x-1\;$ es factor de $\;f(x)=x^{3}-6x^{2}+11x-6.$
		
		\item Verifique si $\;x+2\;$ es factor de $\;f(x)=2x^{3}+3x^{2}-8x-12.$ 
	\end{enumerate}
	
		\bigskip
	
		\noindent\textbf{Instrucciones:} Utilice división sintética
	para obtener el cociente y el residuo

		\item Divida $\;P(x)=x^{5}-4x^{4}+0x^{3}+x^{2}+0x-6\;$ entre $\;x+1.$
		\item Divida $\;P(x)=2x^{3}-5x^{2}+2x+3\;$ entre $\;x-3.$
		\item Divida $\;P(x)=4x^{4}+0x^{3}+0x^{2}-5x+2\;$ entre $\;x-\tfrac{1}{2}.$
	\end{enumerate}
